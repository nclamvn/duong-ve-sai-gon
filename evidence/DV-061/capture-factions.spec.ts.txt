import { test } from '@playwright/test';
import { bootGame, pauseLoop, renderFramesRaf } from './helpers';

for (const pack of ['fictional', 'historical-1971']) {
  test(`capture factions ${pack}`, async ({ page }) => {
    test.setTimeout(300_000);
    await bootGame(page, `factions=${pack}`);
    await pauseLoop(page);
    await page.evaluate(() => {
      const g = window.__ht!.game as unknown as {
        player: { controller: { feet: number[] }; rig: { yaw: number; pitch: number } };
        spawnBot: (id: string, grp: string, p: [number, number, number], o: object) => unknown;
        dummies: Array<{ group: { visible: boolean } }>;
      };
      for (const d of g.dummies) d.group.visible = false;
      const f = g.player.controller.feet;
      const yaw = g.player.rig.yaw;
      const fx = -Math.sin(yaw), fz = -Math.cos(yaw), sx = Math.cos(yaw), sz = -Math.sin(yaw);
      const at = (d: number, s: number): [number, number, number] => [f[0]! + fx * d + sx * s, f[1]!, f[2]! + fz * d + sz * s];
      g.spawnBot('cap_a', 'cap', at(3.2, -0.9), { faction: 'friend' });
      g.spawnBot('cap_b', 'cap', at(3.2, 0.9), { faction: 'enemy' });
      g.player.rig.pitch = -0.12;
    });
    await renderFramesRaf(page, 6);
    await page.screenshot({ path: `evidence/DV-061/phe-${pack}.png` });
  });
}
