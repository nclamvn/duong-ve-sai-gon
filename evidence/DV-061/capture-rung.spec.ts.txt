import { test } from '@playwright/test';
import { renderFramesRaf } from './helpers';

test('capture factions in jungle', async ({ page }) => {
  test.setTimeout(480_000);
  await page.goto('/?backend=webgl&level=truong-son&autostart=1&debug=1&quality=low&shadow=1024');
  await page.waitForFunction(() => window.__ht?.ready === true, null, { timeout: 240_000 });
  await page.waitForFunction(() => (window.__ht?.metrics().frames ?? 0) >= 2, null, { timeout: 240_000 });
  await page.evaluate(() => window.__ht!.game.stop());
  await page.evaluate(() => {
    const g = window.__ht!.game as unknown as {
      player: { controller: { feet: number[] }; rig: { yaw: number; pitch: number } };
      spawnBot: (id: string, grp: string, p: [number, number, number], o: object) => unknown;
      terrain: { heightAt: (x: number, z: number) => number } | null;
    };
    const f = g.player.controller.feet;
    const yaw = g.player.rig.yaw;
    const fx = -Math.sin(yaw), fz = -Math.cos(yaw), sx = Math.cos(yaw), sz = -Math.sin(yaw);
    const at = (d: number, s: number): [number, number, number] => {
      const x = f[0]! + fx * d + sx * s, z = f[2]! + fz * d + sz * s;
      return [x, g.terrain ? g.terrain.heightAt(x, z) : f[1]!, z];
    };
    g.spawnBot('cap_a', 'cap', at(4.5, -1.1), { faction: 'friend' });
    g.spawnBot('cap_b', 'cap', at(6.5, 1.3), { faction: 'enemy' });
    g.player.rig.pitch = -0.06;
  });
  await renderFramesRaf(page, 4);
  await page.screenshot({ path: 'evidence/DV-061/phe-trong-rung.png' });
});
